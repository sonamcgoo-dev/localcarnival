# LCA-0007: Provider Interface Specification

**Status:** Draft  
**Version:** 1.0.0  
**Created:** 2024-01-15

## Overview

The Provider Interface defines how LocalCircus connects to AI model providers. It provides a unified API for chat, completions, embeddings, and other AI capabilities, abstracting away provider-specific details.

## Provider Structure

```typescript
interface AIProvider {
  id: string;
  name: string;
  version: string;
  
  // Capabilities
  capabilities: ProviderCapabilities;
  
  // Models
  models: ModelInfo[];
  
  // API
  chat(options: ChatOptions): Promise<ChatResponse>;
  completion(options: CompletionOptions): Promise<CompletionResponse>;
  embeddings(options: EmbeddingsOptions): Promise<EmbeddingsResponse>;
  
  // Configuration
  configure(options: ProviderOptions): void;
  
  // Health
  health(): Promise<HealthStatus>;
}
```

## Chat API

```typescript
interface ChatOptions {
  model: string;
  messages: ChatMessage[];
  temperature?: number;        // 0.0 - 2.0
  topP?: number;              // 0.0 - 1.0
  maxTokens?: number;         // Max tokens to generate
  stop?: string | string[];    // Stop sequences
  stream?: boolean;            // Enable streaming
  functions?: FunctionDefinition[]; // For function calling
  functionCall?: 'auto' | 'none' | { name: string };
  
  // Provider-specific
  presencePenalty?: number;
  frequencyPenalty?: number;
  logitBias?: Record<string, number>;
  user?: string;
  
  // Metadata
  metadata?: Record<string, string>;
}

interface ChatMessage {
  role: 'system' | 'user' | 'assistant' | 'function';
  content: string;
  name?: string;           // For function messages
  functionCall?: {
    name: string;
    arguments: string;     // JSON string
  };
}

interface ChatResponse {
  model: string;
  choices: ChatChoice[];
  usage: Usage;
  created: number;
  id: string;
  systemFingerprint?: string;
}

interface ChatChoice {
  index: number;
  message: ChatMessage;
  finishReason: 'stop' | 'length' | 'function_call' | 'content_filter' | null;
}

interface Usage {
  promptTokens: number;
  completionTokens: number;
  totalTokens: number;
}
```

## Completion API

```typescript
interface CompletionOptions {
  model: string;
  prompt: string | string[] | ChatPrompt;
  suffix?: string;          // Text after completion
  temperature?: number;
  maxTokens?: number;
  topP?: number;
  n?: number;               // Number of completions
  stream?: boolean;
  logprobs?: number;        // Include log probabilities
  echo?: boolean;           // Echo prompt
  stop?: string | string[];
  
  // Provider-specific
  presencePenalty?: number;
  frequencyPenalty?: number;
  bestOf?: number;
}

type ChatPrompt = Array<{
  role: 'system' | 'user' | 'assistant';
  content: string;
}>;

interface CompletionResponse {
  id: string;
  choices: CompletionChoice[];
  usage: Usage;
  created: number;
  model: string;
}

interface CompletionChoice {
  text: string;
  index: number;
  logprobs?: {
    tokens: string[];
    tokenLogprobs: number[];
    topLogprobs: Record<string, number>[];
    textOffset: number[];
  };
  finishReason: 'stop' | 'length' | 'content_filter' | null;
}
```

## Example Providers

### OpenAI Provider

```typescript
class OpenAIProvider implements AIProvider {
  id = 'openai';
  name = 'OpenAI';
  version = '1.0.0';
  
  capabilities = {
    chat: true,
    completion: true,
    embeddings: true,
    streaming: true,
    functionCalling: true,
    vision: true,
    audio: false,
    jsonMode: true,
    systemPrompt: true,
  };
  
  models = OPENAI_MODELS;
  
  async chat(options: ChatOptions): Promise<ChatResponse> {
    const response = await this.request('/chat/completions', {
      method: 'POST',
      body: {
        model: options.model,
        messages: options.messages,
        temperature: options.temperature,
        max_tokens: options.maxTokens,
        stream: options.stream,
        functions: options.functions,
        function_call: options.functionCall,
      }
    });
    return response;
  }
  
  async embeddings(options: EmbeddingsOptions): Promise<EmbeddingsResponse> {
    const response = await this.request('/embeddings', {
      method: 'POST',
      body: {
        model: options.model,
        input: options.input,
        encoding_format: options.encodingFormat,
        dimensions: options.dimensions,
      }
    });
    return response;
  }
}
```

```typescript
class AnthropicProvider implements AIProvider {
  id = 'anthropic';
  name = 'Anthropic';
  version = '1.0.0';
  
  capabilities = {
    chat: true,
    completion: false,
    embeddings: false,
    streaming: true,
    functionCalling: false,
    vision: true,
    audio: false,
    jsonMode: false,
    systemPrompt: true,
  };
  
  async chat(options: ChatOptions): Promise<ChatResponse> {
    // Convert to Anthropic format
    const response = await this.request('/messages', {
      method: 'POST',
      body: {
        model: options.model,
        messages: options.messages,
        max_tokens: options.maxTokens,
        temperature: options.temperature,
        stream: options.stream,
      }
    });
    // Convert response to standard format
    return this.convertResponse(response);
  }
}
```
