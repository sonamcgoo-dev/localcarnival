# LCA-0006: Compatibility Matrix Specification

**Status:** Draft  
**Version:** 1.0.0  
**Created:** 2024-01-15

## Overview

The Compatibility Matrix defines how artifacts interact with platforms, architectures, and each other. It provides the framework for resolving dependencies and ensuring artifacts work correctly in the user's environment.

## Compatibility Dimensions

### Platform Compatibility

Artifacts declare which platforms they support:

```typescript
type Platform = 'windows' | 'macos' | 'linux' | 'wasm';

interface PlatformCompatibility {
  platforms: Platform[];
  exceptions?: {
    platform: Platform;
    reason: string;
    versionRange?: string;
  }[];
}
```

### Architecture Compatibility

```typescript
type Architecture = 'x64' | 'arm64' | 'wasm32';

interface ArchitectureCompatibility {
  architectures: Architecture[];
  optimizations?: {
    architecture: Architecture;
    implementation: string; // e.g., 'avx2', 'neon'
  }[];
}
```

### LocalCircus Version Compatibility

```typescript
interface VersionRange {
```


## Resolution Algorithm

### Compatibility Check

```typescript
function checkCompatibility(
  artifact: ArtifactDNA,
  environment: Environment
): CompatibilityResult {
  const result: CompatibilityResult = {
    compatible: true,
    issues: []
  };

  // Check platform
  if (artifact.compatibility.platforms?.length) {
    if (!artifact.compatibility.platforms.includes(environment.platform)) {
      result.compatible = false;
      result.issues.push({
        type: 'platform',
        message: `Requires platform: ${artifact.compatibility.platforms.join(', ')}`,
        severity: 'error'
      });
    }
  }

  // Check architecture
  if (artifact.compatibility.architectures?.length) {
    if (!artifact.compatibility.architectures.includes(environment.architecture)) {
      result.compatible = false;
      result.issues.push({
        type: 'architecture',
        message: `Requires architecture: ${artifact.compatibility.architectures.join(', ')}`,
        severity: 'error'
      });
    }
  }

  // Check LocalCircus version
  if (artifact.compatibility.localcircus) {
    const lcVersion = environment.localcircusVersion;
    
    if (artifact.compatibility.localcircus.min) {
      if (!semver.gte(lcVersion, artifact.compatibility.localcircus.min)) {
        result.compatible = false;
        result.issues.push({
          type: 'version',
          message: `Requires LocalCircus ${artifact.compatibility.localcircus.min}+`,
          severity: 'error'
        });
      }
    }
    
    if (artifact.compatibility.localcircus.max) {
      if (!semver.lte(lcVersion, artifact.compatibility.localcircus.max)) {
        result.compatible = false;
        result.issues.push({
          type: 'version',
          message: `Not compatible with LocalCircus ${lcVersion}`,
          severity: 'error'
        });
      }
    }
  }

  return result;
}

interface Environment {
  platform: Platform;
  architecture: Architecture;
  localcircusVersion: string;
  features: string[];
}
```

### Dependency Resolution

```typescript
interface DependencyResolution {
  // Resolved dependencies
  resolved: Map<string, ResolvedDependency>;
  
  // Unresolved dependencies
  unresolved: UnresolvedDependency[];
  
  // Conflicts
  conflicts: DependencyConflict[];
}

interface ResolvedDependency {
  name: string;
  requestedVersion: string;
  resolvedVersion: string;
  artifact: ArtifactDNA;
  path: string[]; // Resolution path
}

interface UnresolvedDependency {
  name: string;
  requestedVersion: string;
  reason: string;
}

interface DependencyConflict {
  name: string;
  versions: string[];
  reason: string;
}

function resolveDependencies(
  artifact: ArtifactDNA,
  registry: Registry,
  environment: Environment
): DependencyResolution {
  const resolved = new Map<string, ResolvedDependency>();
  const unresolved: UnresolvedDependency[] = [];
  const conflicts: DependencyConflict[] = [];
  
  // Use topological sort with version constraints
  const queue: Array<{ name: string; versionRange: string; path: string[] }> = [];
  
  for (const dep of artifact.dependencies) {
    queue.push({
      name: dep.reference.name,
      versionRange: dep.versionRange,
      path: [artifact.name]
    });
  }
  
  while (queue.length > 0) {
    const current = queue.shift()!;
    
    if (resolved.has(current.name)) continue;
    
    const candidates = registry.findByName(current.name);
    
    if (candidates.length === 0) {
      unresolved.push({
        name: current.name,
        requestedVersion: current.versionRange,
        reason: 'Not found in registry'
      });
      continue;
    }
    
    // Find best matching version
    const matching = candidates
      .filter(c => semver.satisfies(c.version, current.versionRange))
      .filter(c => checkCompatibility(c, environment).compatible);
    
    if (matching.length === 0) {
      unresolved.push({
        name: current.name,
        requestedVersion: current.versionRange,
        reason: 'No compatible version found'
      });
      continue;
    }
    
    // Pick highest compatible version
    const best = matching.sort((a, b) => semver.rcompare(a.version, b.version))[0];
    
    resolved.set(current.name, {
      name: current.name,
      requestedVersion: current.versionRange,
      resolvedVersion: best.version,
      artifact: best,
      path: current.path
    });
    
    // Add transitive dependencies
    for (const dep of best.dependencies) {
      queue.push({
        name: dep.reference.name,
        versionRange: dep.versionRange,
        path: [...current.path, current.name]
      });
    }
  }
  
  return { resolved, unresolved, conflicts };
}
```

## Capability Requirements

```typescript
interface CapabilityRequirement {
  capabilityId: string;
  minVersion?: string;
  optional: boolean;
}

// Type-specific requirements
const TYPE_REQUIREMENTS: Record<ArtifactType, CapabilityRequirement[]> = {
  'model': [
    { capabilityId: 'chat', optional: false },
  ],
  'agent': [
    { capabilityId: 'chat', optional: false },
    { capabilityId: 'streaming', optional: true },
  ],
  'workflow': [],
  'tool': [],
  'plugin': [
    { capabilityId: 'event-bus', optional: false },
  ],
  'mcp-server': [
    { capabilityId: 'http', optional: false },
  ],
};

function checkCapabilityRequirements(
  artifact: ArtifactDNA,
  capabilities: Capability[]
): CapabilityResult {
  const requirements = TYPE_REQUIREMENTS[artifact.type] || [];
  const missing: string[] = [];
  
  for (const req of requirements) {
    const found = capabilities.find(
      c => c.id === req.capabilityId &&
           (!req.minVersion || semver.satisfies(c.version || '1.0.0', `>=${req.minVersion}`))
    );
    
    if (!found && !req.optional) {
      missing.push(req.capabilityId);
    }
  }
  
  return {
    satisfied: missing.length === 0,
    missing: missing,
    warnings: requirements
      .filter(r => r.optional && !capabilities.some(c => c.id === r.capabilityId))
      .map(r => `Optional capability not available: ${r.capabilityId}`)
  };
}
```
